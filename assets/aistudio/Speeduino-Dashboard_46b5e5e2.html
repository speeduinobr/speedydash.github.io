1 Visão Geral do Projeto

1.1 Objetivo Principal

Desenvolver um sistema de dashboard ECU multi-plataforma (Desktop Linux + Android Mobile) com:

  • Visual moderno e altamente customizável
  • Múltiplos temas trocáveis em tempo real
  • Performance gráfica de alta velocidade (60-120fps)
  • Conexão Bluetooth com ECUs (Speeduino, MegaSquirt, etc.)
  • Gauges analógicos fluidos e charts em tempo real

1.3 Stack Tecnológica Final

Frontend Core: ├─ React 18+ com TypeScript ├─ Vite (build tool) ├─ Tailwind CSS (estilização) ├─ Zustand (state management) └─ CSS Variables (sistema de temas) Renderização Gráfica: ├─ Canvas 2D API (gauges principais) ├─ WebGL/Three.js (efeitos glow/shaders) ├─ OffscreenCanvas + Workers (performance) └─ Lightweight Charts / ECharts (waveforms) Conectividade: ├─ Web Bluetooth API (comunicação ECU) └─ WebSocket fallback (se necessário) Desktop Wrapper: └─ Tauri 2.0 (Rust + WebView2) Mobile: └─ PWA (Progressive Web App)

interface ECUState {
  // Dados dos sensores
  sensors: {
    rpm: number;          // 0-8000+ RPM
    map: number;          // 0-250 kPa
    tps: number;          // 0-100 %
    clt: number;          // -40 a 130 °C
    iat: number;          // -40 a 80 °C
    lambda: number;       // 0.6 a 1.4 λ
    wue: number;          // 0-100 % (warmup)
    advance: number;      // 0-60 ° (ignition advance)
    battery: number;      // 10-15V
    afr: number;          // Air-Fuel Ratio
  };
  
  // Status do motor
  status: {
    warmup: boolean;
    cranking: boolean;
    halfSync: boolean;
    ase: boolean;
    syncLoss: number;
    engineRunning: boolean;
  };
  
  // Conexão Bluetooth
  bluetooth: {
    connected: boolean;
    device: BluetoothDevice | null;
    rssi: number;
    lastUpdate: number;
  };
  
  // Configurações
  config: {
    activeTheme: ThemeType;
    refreshRate: number;    // 30 ou 60 Hz
    layout: LayoutType;
    gauges: GaugeConfig[];
  };
}

// Ring buffer para performance
class DataBuffer {
  private buffer: Float32Array;
  private index: number = 0;
  
  push(value: number) {
    this.buffer[this.index] = value;
    this.index = (this.index + 1) % this.buffer.length;
  }
  
  getData(): Float32Array {
    return this.buffer;
  }
}

4.4 Bluetooth - Web Bluetooth API

export function useBluetooth() {
  const [connected, setConnected] = useState(false);
  const [data, setData] = useState<ECUData>({
    rpm: 0, map: 0, tps: 0, clt: 0, iat: 0, lambda: 0
  });

  const connect = async () => {
    try {
      const device = await navigator.bluetooth.requestDevice({
        filters: [{ namePrefix: 'Speeduino' }],
        optionalServices: ['0000ffe0-0000-1000-8000-00805f9b34fb']
      });

      const server = await device.gatt.connect();
      const service = await server.getPrimaryService(
        '0000ffe0-0000-1000-8000-00805f9b34fb'
      );
      const characteristic = await service.getCharacteristic(
        '0000ffe1-0000-1000-8000-00805f9b34fb'
      );

      await characteristic.startNotifications();
      characteristic.addEventListener(
        'characteristicvaluechanged', 
        (event) => {
          const data = new DataView(
            event.target.value.buffer
          );
          setData({
            rpm: data.getUint16(0, true),
            map: data.getUint16(2, true) / 10,
            tps: data.getUint8(4),
            clt: data.getInt8(5),
            iat: data.getInt8(6),
            lambda: data.getUint16(7, true) / 1000,
          });
        }
      );

      setConnected(true);
    } catch (error) {
      console.error('Bluetooth error:', error);
    }
  };

  return { connected, data, connect };
}
